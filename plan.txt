Query Payload Structure:

GraphQL
query submissionList($offset: Int!, $limit: Int!, $slug: String) {
  submissionList(offset: $offset, limit: $limit, questionSlug: $slug) {
    hasNext
    submissions {
      id
      statusDisplay
      lang
      timestamp
    }
  }
}
